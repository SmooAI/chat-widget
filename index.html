@smooai/chat-widget

Embeddable chat widget for smooth-operator-agent. The launcher in the bottom-right is a <smooth-agent-chat> custom element wired to the @smooai/smooth-operator-agent protocol client. Point endpoint at a running smooth-operator-agent WebSocket service and set a real agent-id.

Declarative embed

<script src="dist/chat-widget.global.js"></script>
<smooth-agent-chat
    endpoint="ws://localhost:8787/ws"
    agent-id="00000000-0000-0000-0000-000000000000"
    agent-name="Support"
    start-open>
</smooth-agent-chat>

Programmatic embed

<script src="dist/chat-widget.global.js"></script>
<script>
    SmoothAgentChat.mount({
        endpoint: 'ws://localhost:8787/ws',
        agentId: '00000000-0000-0000-0000-000000000000',
        agentName: 'Support',
        theme: { primary: '#7c3aed' },
    });
</script>
